### **2. Script Manager (Lua Engine)**

```cpp
class ScriptManager {
private:
    lua_State* m_luaState;
    std::unordered_map<std::string, std::string> m_scriptCache;
    
public:
    void initialize() {
        m_luaState = luaL_newstate();
        luaL_openlibs(m_luaState);
        
        // Expose C++ functions to Lua
        exposeEngineAPI();
    }
    
    void exposeEngineAPI() {
        // Expose transform functions
        lua_register(m_luaState, "setPosition", lua_setPosition);
        lua_register(m_luaState, "getPosition", lua_getPosition);
        lua_register(m_luaState, "moveTowards", lua_moveTowards);
        
        // Expose game functions
        lua_register(m_luaState, "findNearbyPlayers", lua_findNearbyPlayers);
        lua_register(m_luaState, "playSound", lua_playSound);
        lua_register(m_luaState, "changeState", lua_changeState);
    }
    
    void reloadScript(const std::string& scriptPath) {
        // Hot-reload script while game runs
        if (m_scriptCache.find(scriptPath) != m_scriptCache.end()) {
            executeScript(scriptPath);
        }
    }
};
```

```cpp
class ScriptSystem : public System {
public:
    void update(entt::registry& registry, float deltaTime) override {
        auto view = registry.view<Script, Transform>();
        
        for (auto entity : view) {
            auto [script, transform] = view.get<Script, Transform>(entity);
            
            if (script.isEnabled && script.luaState) {
                // Call Lua update function
                lua_getglobal(script.luaState, "update");
                lua_pushnumber(script.luaState, deltaTime);
                lua_pcall(script.luaState, 1, 0, 0);
            }
        }
    }
};
```

```lua
    self.speed = 5.0
    self.state = "patrol"
    self.patrolPoints = {
        {x = 0, y = 0, z = 0},
        {x = 10, y = 0, z = 0},
        {x = 10, y = 0, z = 10},
        {x = 0, y = 0, z = 10}
    }
    self.currentPoint = 1
end

function NPC:update(deltaTime)
    if self.state == "patrol" then
        self:patrol(deltaTime)
    elseif self.state == "chase" then
        self:chasePlayer(deltaTime)
    elseif self.state == "attack" then
        self:attack(deltaTime)
    end
end

function NPC:patrol(deltaTime)
    local targetPoint = self.patrolPoints[self.currentPoint]
    local distance = self:distanceTo(targetPoint)
    
    if distance < 1.0 then
        self.currentPoint = (self.currentPoint % #self.patrolPoints) + 1
    else
        self:moveTowards(targetPoint, deltaTime)
    end
end

function NPC:onPlayerDetected(player)
    self.target = player
    self.state = "chase"
    playSound("alert")
end

return NPC
```

```lua
    self.fireRate = 0.1
    self.lastFireTime = 0
    self.ammo = 12
    self.maxAmmo = 12
end

        self:reload()
    end
end

function Weapon:fire()
    local currentTime = getTime()
    if currentTime - self.lastFireTime >= self.fireRate and self.ammo > 0 then
        self:createBullet()
        self.ammo = self.ammo - 1
        self.lastFireTime = currentTime
        playSound("gunshot")
    end
end

function Weapon:reload()
    self.ammo = self.maxAmmo
    playSound("reload")
end

return Weapon
```

```lua
    self.maxPlayers = 8
    self.timeLimit = 300
    self.currentTime = 0
    self.players = {}
    self.targets = {}
    self.scores = {}
end

    if self.currentTime >= self.timeLimit then
        self:endGame()
    end
    
    self:updateTargets(deltaTime)
    self:checkPlayerCollisions()
end

function Arena:onPlayerJoin(player)
    table.insert(self.players, player)
    self.scores[player.id] = 0
    player:setSpawnPoint(self:getRandomSpawnPoint())
end

function Arena:onTargetDestroyed(target, player)
    local score = target.scoreValue
    self.scores[player.id] = self.scores[player.id] + score
    player:showScorePopup(score)
end

return Arena
```
